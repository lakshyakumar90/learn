## Debounced search

Type tea quickly. Only the final query should run, and cancellation should stop pending work.

<Diagram id="js-runtime" />

<CodeRunner title="Build along: Debounced search" expect="tea\nrequests 1">
{`function debounce(callback, delay) {
  let timer;
  function schedule(...args) {
    clearTimeout(timer);
    timer = setTimeout(() => callback(...args), delay);
  }
  schedule.cancel = () => clearTimeout(timer);
  return schedule;
}
const queries = [];
const search = debounce((query) => queries.push(query), 5);
search("t");
search("te");
search("tea");
await sleep(15);
console.log(queries.join(","));
search("coffee");
search.cancel();
await sleep(15);
console.log("requests", queries.length);`}
</CodeRunner>

## To-do list

Reject blank items, create safe text nodes and delegate actions to the list.

<Diagram id="js-event-flow" />

<CodeRunner title="Build along: To-do list" expect="true\nfalse\nbuy tea\ncount 1\ncontract true">
{`function makeTodoStore() {
  let todos = [];
  return {
    add(text) {
      const clean = text.trim();
      if (!clean) return false;
      todos = [...todos, { id: todos.length + 1, text: clean }];
      return true;
    },
    list() {
      return todos.map((todo) => ({ ...todo }));
    },
  };
}
const store = makeTodoStore();
console.log(store.add(" buy tea "));
console.log(store.add(""));
console.log(store.list()[0].text);
console.log("count", store.list().length);
const contractChecks = [
  store.list().length === 1,
  store.list()[0].text === "buy tea",
];
if (!contractChecks.every(Boolean)) {
  throw new Error("The boundary contract was broken");
}
console.log("contract", contractChecks.every(Boolean));`}

## Form validation

Accept positive integer quantities and reject blanks, zero and mixed text before calculating money.

<Diagram id="js-types" />

<CodeRunner title="Build along: Form validation" expect="pence 250\ninvalid &quot;&quot;\ninvalid &quot;2x&quot;\ninvalid &quot;0&quot;\npence 375\nfallback preserves zero 0\ncontract true">
{`function parseQuantity(input) {
  if (typeof input !== "string" || input.trim() === "") return null;
  const quantity = Number(input);
  if (!Number.isInteger(quantity) || quantity < 1) return null;
  return quantity;
}
const priceInPence = 125;
const inputs = ["2", "", "2x", "0", "3"];
for (const input of inputs) {
  const quantity = parseQuantity(input);
  if (quantity === null) {
    console.log("invalid", JSON.stringify(input));
    continue;
  }
  const total = quantity * priceInPence;
  console.log("pence", total);
}
console.log("fallback preserves zero", 0 ?? 10);
const contractChecks = [parseQuantity("") === null, parseQuantity("2") === 2];
if (!contractChecks.every(Boolean)) {
  throw new Error("The boundary contract was broken");
}
console.log("contract", contractChecks.every(Boolean));`}
</CodeRunner>

## Infinite scroll

Consume one page at a time, keep a cursor and cancel when the view disappears.

<Diagram id="js-runtime" />

<CodeRunner title="Build along: Infinite scroll" expect="first 1,2,3\nsecond 4,5,6\nlast 7\ndone true\nbatch size 2\nbatch size 2\nbatch size 2\nbatch size 1">
{`function* pages(records, size) {
  for (let start = 0; start < records.length; start += size) {
    yield records.slice(start, start + size);
  }
}
const records = Array.from({ length: 7 }, (_, index) => ({
  id: index + 1,
}));
const feed = pages(records, 3);
console.log(
  "first",
  feed
    .next()
    .value.map((row) => row.id)
    .join(","),
);
console.log(
  "second",
  feed
    .next()
    .value.map((row) => row.id)
    .join(","),
);
console.log(
  "last",
  feed
    .next()
    .value.map((row) => row.id)
    .join(","),
);
console.log("done", feed.next().done);
const freshFeed = pages(records, 2);
for (const batch of freshFeed) console.log("batch size", batch.length);`}
</CodeRunner>

## Fetch with retry and timeout

Overlap independent reads, bound retry attempts and clear deadline handles.

<Diagram id="js-promise-states" />

<CodeRunner title="Build along: Fetch with retry and timeout" expect="Aarav Sharma\nposts 3">
{`async function loadJson(url, attempts = 3) {
  for (let attempt = 1; attempt <= attempts; attempt++) {
    const controller = new AbortController();
    const timer = setTimeout(() => controller.abort(), 100);
    try {
      const response = await fakeFetch(url, {
        latencyMs: 5,
        signal: controller.signal,
      });
      if (!response.ok) throw new Error("HTTP " + response.status);
      return await response.json();
    } catch (error) {
      if (attempt === attempts) throw error;
      await sleep(5 * attempt);
    } finally {
      clearTimeout(timer);
    }
  }
}
const [users, posts] = await Promise.all([
  loadJson("/users"),
  loadJson("/posts"),
]);
console.log(users[0].name);
console.log("posts", posts.length);`}

## Event bus

Publish paid, subscribe a receipt handler and verify that unsubscription stops another receipt.

<Diagram id="js-event-flow" />

<CodeRunner title="Build along: Event bus" expect="receipt 42\nsubscription released">
{`function makeBus() {
  const listeners = new Map();
  return {
    on(event, handler) {
      const group = listeners.get(event) ?? new Set();
      group.add(handler);
      listeners.set(event, group);
      return () => group.delete(handler);
    },
    emit(event, value) {
      for (const handler of listeners.get(event) ?? []) handler(value);
    },
  };
}
const bus = makeBus();
const unsubscribe = bus.on("paid", (id) => console.log("receipt", id));
bus.emit("paid", 42);
unsubscribe();
bus.emit("paid", 43);
console.log("subscription released");`}
</CodeRunner>

## Rate limiter

Give accounts independent attempt state and reject calls after the limit.

<Diagram id="js-scope-chain" />

<CodeRunner title="Build along: Rate limiter" expect="true\ntrue\nfalse\ntrue\nfalse">
{`function makeLimiter(limit) {
  let remaining = limit;
  return () => {
    if (remaining === 0) return false;
    remaining--;
    return true;
  };
}
const login = makeLimiter(2);
const search = makeLimiter(1);
console.log(login());
console.log(login());
console.log(login());
console.log(search());
console.log(search());`}
</CodeRunner>
