### Build the working version

<CodeRunner title="Build: JIT compilation and performance" expect="total 12\ntotal 12\ntotal 12\nconsistent results matter more than guessed speed\nvalid rows 2\ncontract true">
{`function sumPrices(rows) {
  let total = 0;
  for (const row of rows) {
    const price = Number(row.price);
    if (!Number.isFinite(price)) continue;
    total += price;
  }
  return total;
}
const incoming = [{ price: "4" }, { price: 8 }, { price: "invalid" }];
const normalized = incoming.map((row) => ({ price: Number(row.price) }));
for (let pass = 0; pass < 3; pass++)
  console.log("total", sumPrices(normalized));
console.log("consistent results matter more than guessed speed");
console.log(
  "valid rows",
  normalized.filter((row) => Number.isFinite(row.price)).length,
);
const contractChecks = [sumPrices(normalized) === 12, normalized.length === 3];
if (!contractChecks.every(Boolean)) {
  throw new Error("The boundary contract was broken");
}
console.log("contract", contractChecks.every(Boolean));`}
</CodeRunner>

## Why it works this way

Normalization also costs time. Profile the complete action and compare representative warmed workloads rather than isolated cold runs.
