<Exercise
  title="2. cart reducer"
  language="typescript"
  tests={`assert(reduceCart(10, { kind: "add", amount: 5 }) === 15, "adds");
assert(reduceCart(10, { kind: "reset" }) === 0, "resets");`}
  solution={`type CartAction = { kind: "add"; amount: number } | { kind: "reset" };
function reduceCart(total: number, action: CartAction): number {
  if (action.kind === "add") {
    return total + action.amount;
  }
  return 0;
}`}
>
{`type CartAction = { kind: "add"; amount: number } | { kind: "reset" };
function reduceCart(total: number, action: CartAction): number {
  if (action.kind === "add") {
    return total + action.amount;
  }
  return total + action.amount;
}`}
</Exercise>

<Exercise
  title="3. safe display name"
  language="typescript"
  tests={`assert(displayName({ name: "sam" }) === "sam", "reads name");
assert(displayName({}) === "guest", "missing name");
assert(displayName(null) === "guest", "null");
assert(displayName("sam") === "guest", "wrong shape");`}
  solution={`function displayName(raw: unknown): string {
  if (typeof raw === "object" && raw !== null && "name" in raw) {
    const record = raw as { name: unknown };
    if (typeof record.name === "string") {
      return record.name;
    }
  }
  return "guest";
}`}
>
{`function displayName(raw: unknown): string {
  const record = raw as { name: string };
  return record.name;
}`}
</Exercise>

<Exercise
  title="4. typed event bus"
  language="typescript"
  tests={`const heard: string[] = [];
const bus = new Bus();
bus.on(function (message: string) {
  heard.push(message);
});
bus.emit("hi");
bus.emit("there");
assert(heard.length === 2, "both heard");
assert(heard.join(",") === "hi,there", "in order");`}
  solution={`type Handler = (message: string) => void;
class Bus {
  private handlers: Handler[] = [];
  on(handler: Handler): void {
    this.handlers.push(handler);
  }
  emit(message: string): void {
    for (const handler of this.handlers) {
      handler(message);
    }
  }
}`}
>
{`type Handler = (message: string) => void;
class Bus {
  private handlers: unknown[] = [];
  on(handler: Handler): void {
    this.handlers.push(handler);
  }
  emit(message: string): void {
    for (const handler of this.handlers) {
      handler(message);
    }
  }
}`}
</Exercise>

<Exercise
  title="8. rename without mutating"
  language="typescript"
  tests={`const cart: CartItem[] = [
  { id: 1, name: "book", price: 12 },
  { id: 2, name: "pen", price: 3 },
];
const next = rename(cart, 1, "novel");
assert(next[0].name === "novel", "renamed");
assert(cart[0].name === "book", "original kept");
assert(next[1] === cart[1], "others shared");`}
  solution={`interface CartItem {
  readonly id: number;
  name: string;
  price: number;
}
function rename(items: CartItem[], id: number, name: string): CartItem[] {
  return items.map(function (item) {
    if (item.id === id) {
      return { id: item.id, name: name, price: item.price };
    }
    return item;
  });
}`}
>
{`interface CartItem {
  readonly id: number;
  name: string;
  price: number;
}
function rename(items: CartItem[], id: number, name: string): CartItem[] {
  for (const item of items) {
    if (item.id === id) {
      item.name = name;
    }
  }
  return items;
}`}
</Exercise>
